make_control_groups <- function(df, 
                                ever_added,
                                ever_removed,
                    pre_eff_change_type, 
                    type = c("pre", "eff")
                                ){
  
  # Treated Group A: Added to Risk Zone
  group_T_A <- df |>
    filter( {{ever_added}} == TRUE )
  
  # Treated Group B: Removed from Risk Zone
  group_T_B <- df |> 
      filter( {{ever_removed}}  == TRUE)

  # Control Group A:  Never in SFHA, FIRM panels never updated
    
   if(type == "pre"){
   group_C_A <- df |> 
      filter(
        {{pre_eff_change_type}} == "Never SFHA" &
             pre_date_chr %in% c("2005-01-01")
         )
  }else if(type == "eff"){  # for effective maps
  group_C_A <- df |> 
      filter(
        {{pre_eff_change_type}} == "Never SFHA" &
         eff_date_chr %in% c("2008-08-19") 
         ) 
  }  

    # Control Group B: Never in SFHA, FIRM Panels are updated
    group_C_B  <- df |> 
      filter( 
        {{pre_eff_change_type}} == "Never SFHA" &
         (eff_date_chr %in% c("2019-11-01", "2021-09-10") | 
             pre_date_chr %in% c("2015-02-12", "2019-07-01", "2021-09-22")
         )
         )

    # Control Group C: Always in SFHA, FIRM Panels are not updated 
    if(type == "pre"){
    group_C_C <- df |> 
      filter(
        {{pre_eff_change_type}} == "Always SFHA" &             
          pre_date_chr %in% c("2005-01-01")
      )}
    else if(type == "eff") {  # this probably is tainted by prelim map cohort in 2021 though....
    group_C_C <- df |> 
      filter(
        {{pre_eff_change_type}} == "Always SFHA" &
         (eff_date_chr %in% c("2008-08-19") )
      )
    
    group_C_C_alt <- df |> 
      filter(
        {{pre_eff_change_type}} == "Always SFHA" &
         (eff_date_chr %in% c("2008-08-19") | 
            pre_date_chr %in% c("2005-01-01")
          )
      )
    }
    
    # Always in SFHA but FIRM panels have been recently updated
   
    if(type == "eff") {
      group_C_D <- df |> 
        filter(
        {{pre_eff_change_type}} == "Always SFHA" &
         (eff_date_chr %in% c("2019-11-01", "2021-09-10"))
         )
      
       group_C_D_alt <- df |> 
        filter(
        {{pre_eff_change_type}} == "Always SFHA" &
         (eff_date_chr %in% c("2019-11-01", "2021-09-10") | 
             pre_date_chr %in% c("2015-02-12", "2019-07-01", "2021-09-22")
             )
         )
    } else if (type == "pre") {
      group_C_D <- df |> 
         filter(
        {{pre_eff_change_type}} == "Always SFHA" &
             pre_date_chr %in% c("2015-02-12", "2019-07-01", "2021-09-22")
         )
    }
      list(
    group_T_A = group_T_A,
    group_T_B = group_T_B,
    group_C_A = group_C_A,
    group_C_B = group_C_B,
    group_C_C = group_C_C,
    group_C_D = group_C_D

  )

}

Additions: event study and pretrend tests

Properties added to the preliminary SFHA are compared with never-SFHA properties. Event time is relative to the preliminary map year; sales earlier than the preliminary date within that year are assigned to year -1. The endpoint bins pool years at or below -5 and at or above +5. Two-year-early placebo tests

Appendix A discusses whether estimated changes appear before the preliminary map date. These placebo models move each preliminary date two years earlier, restrict observations to sales strictly before the actual preliminary date, and estimate the treated-by-placebo-period interaction with property and year fixed effects and parcel-clustered standard errors. Removals are displayed first, then additions.

A nonzero placebo estimate can reflect anticipation, differential trends, or other misspecification; it does not identify which explanation is responsible. A non-significant estimate does not rule these out. The final two tables use the actual preliminary date and the broader time window, providing a reference specification rather than additional placebo tests.

Code 
# df_prep must have: pin, pin10, sale_date, sale_year, pre_date, prelim_sfha_category, log_price

run_placebo_did <- function(df,
                            fake_shift_years = 2,
                            treated_label = "Added to prelim SFHA",
                            unit_fe = "pin",
                            time_fe = "sale_year",
                            cluster_var = "pin10") {
  if(treated_label == "Added to prelim SFHA"){
    df <- df |> filter(prelim_sfha_category %in% c("Added to prelim SFHA", "Never SFHA"))
  }
  if(treated_label == "Removed from prelim SFHA"){
    df <- df |> filter(prelim_sfha_category %in% c("Removed from prelim SFHA", "Always SFHA"))
  }
  df_placebo <- df |>
    mutate(
      treated = prelim_sfha_category == treated_label,
      placebo_date = pre_date %m-% years(fake_shift_years),
      placebo_post = sale_date >= placebo_date
    ) |>
    # keep only observations strictly before the *real* event (so placebo is truly placebo)
    filter(sale_date < pre_date) |>
    # optional: keep only observations with a defined placebo_date
    filter(!is.na(placebo_date))

  fml <- as.formula(paste0(
    "log_price ~ treated * placebo_post | ",
    unit_fe, " + ", time_fe
  ))

  feols(
    fml,
    cluster = as.formula(paste0("~", cluster_var)),
    data = df_placebo
  )
}

Regional controls and conditional parallel trends

Appendix A considers whether regional housing-market differences affect the Q1 comparisons. The specifications below adapt the existing triad-by-year event studies in q1_assumption_tests.qmd to the same addition and removal samples used above. They replace common year effects with sale_year^Triad while retaining property fixed effects and parcel-clustered standard errors. These are region-by-year fixed effects, not separate linear trend slopes.

The comparisons use a common complete-case sample for each treatment direction, so differences between the two specifications are not caused by missing triad identifiers. The joint tests concern the included pretreatment coefficients relative to year -1; failure to reject does not establish the unobserved post-treatment counterfactual. These tables are newly assembled from the existing specifications, not a verified reproduction of a numbered dissertation table.

```{r}
  d <- readRDS(path) |>
    dplyr::filter(!pin10 %in% drop_parcels | (pin10 == "0113301013" & sale_year > 2014))

    `Preliminary: added` = "addedto_prelim_sfha",
    `Preliminary: removed` = "removedfrom_prelim_sfha",
    `Effective: added` = "addedto_eff_sfha",
    `Effective: removed` = "removedfrom_eff_sfha"
  )
  if (!all(c("pin10", unname(indicators)) %in% names(d))) {
    stop("Missing transition identifiers in ", path)
  }

    flag <- as.character(d[[variable]])

    }
    treated <- !is.na(flag) & flag %in% c("TRUE", "1")
    tibble::tibble(
      Definition = definition, Transition = transition,
      `Input sales` = nrow(d), `Treated sales` = sum(treated),
      `Treated parcels` = dplyr::n_distinct(d$pin10[treated], na.rm = TRUE),
      `Missing transition flag` = sum(is.na(flag))
    )
  })
```

## Two-year-early placebo tests {#q1-placebo-tests}

Appendix A discusses whether estimated changes appear before the preliminary map date. These placebo models move each preliminary date two years earlier, restrict observations to sales strictly before the actual preliminary date, and estimate the treated-by-placebo-period interaction with property and year fixed effects and parcel-clustered standard errors. Removals are displayed first, then additions.

A nonzero placebo estimate can reflect anticipation, differential trends, or other misspecification; it does not identify which explanation is responsible. A non-significant estimate does not rule these out. The final two tables use the actual preliminary date and the broader time window, providing a reference specification rather than additional placebo tests.


```{r}
#| label: tbl-appendixA-placebotest
#| 

# df_prep must have: pin, pin10, sale_date, sale_year, pre_date, prelim_sfha_category, log_price

run_placebo_did <- function(df,
                            fake_shift_years = 2,
                            treated_label = "Added to prelim SFHA",
                            unit_fe = "pin",
                            time_fe = "sale_year",
                            cluster_var = "pin10") {
  if(treated_label == "Added to prelim SFHA"){
    df <- df |> filter(prelim_sfha_category %in% c("Added to prelim SFHA", "Never SFHA"))
  }
  if(treated_label == "Removed from prelim SFHA"){
    df <- df |> filter(prelim_sfha_category %in% c("Removed from prelim SFHA", "Always SFHA"))
  }
  df_placebo <- df |>
    mutate(
      treated = prelim_sfha_category == treated_label,
      placebo_date = pre_date %m-% years(fake_shift_years),
      placebo_post = sale_date >= placebo_date
    ) |>
    # keep only observations strictly before the *real* event (so placebo is truly placebo)
    filter(sale_date < pre_date) |>
    # optional: keep only observations with a defined placebo_date
    filter(!is.na(placebo_date))

  fml <- as.formula(paste0(
    "log_price ~ treated * placebo_post | ",
    unit_fe, " + ", time_fe
  ))

  feols(
    fml,
    cluster = as.formula(paste0("~", cluster_var)),
    data = df_placebo
  )
}
```

## Regional controls and conditional parallel trends {#q1-conditional-trends}

Appendix A considers whether regional housing-market differences affect the Q1 comparisons. The specifications below adapt the existing triad-by-year event studies in `q1_assumption_tests.qmd` to the same addition and removal samples used above. They replace common year effects with `sale_year^Triad` while retaining property fixed effects and parcel-clustered standard errors. These are region-by-year fixed effects, not separate linear trend slopes.

The comparisons use a common complete-case sample for each treatment direction, so differences between the two specifications are not caused by missing triad identifiers. The joint tests concern the included pretreatment coefficients relative to year -1; failure to reject does not establish the unobserved post-treatment counterfactual. These tables are newly assembled from the existing specifications, not a verified reproduction of a numbered dissertation table.
